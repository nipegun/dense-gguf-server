cmake_minimum_required(VERSION 3.19)
project(dgs VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BUILD_SHARED_LIBS "Build shared runtime libraries" OFF)
option(DGS_BUILD_UI "Build and embed the associated web application" ON)
option(DGS_LTO "Enable link-time optimization" ON)
option(DGS_OPENSSL "Enable HTTPS" ON)
option(DGS_SUBPROCESS "Enable server tools, router and video subprocesses" ON)
option(DGS_ALL_WARNINGS "Enable compiler warnings" ON)
option(DGS_FATAL_WARNINGS "Treat warnings as errors" OFF)
option(DGS_LLGUIDANCE "Enable optional LLGuidance structured output" OFF)

set(DGS_BUILD_COMMON ON)
set(DGS_TOOLS_INSTALL ON)
set(DGS_VERSION_BASE "${PROJECT_VERSION}")
set(DGS_VERSION_MAJOR "${PROJECT_VERSION_MAJOR}")
set(DGS_VERSION "${PROJECT_VERSION}")
set(DGS_BUILD_NUMBER 0 CACHE STRING "Release build number")
set(DGS_BUILD_COMMIT "dense-only")
set(BUILD_NUMBER "${DGS_BUILD_NUMBER}")
set(BUILD_COMMIT "${DGS_BUILD_COMMIT}")
set(BUILD_COMPILER "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
set(BUILD_TARGET "${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_PROCESSOR}")
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_ALL_WARNINGS "${DGS_ALL_WARNINGS}")
set(GGML_CUDA_GRAPHS_DEFAULT ON)
set(GGML_DGSFILE_DEFAULT ON)

if(DGS_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT vIpoSupported OUTPUT vIpoError)
  if(vIpoSupported)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  else()
    message(STATUS "LTO unavailable: ${vIpoError}")
  endif()
endif()
if(WIN32)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
if(MSVC)
  add_compile_options(/utf-8 /bigobj)
endif()
include(GNUInstallDirs)
include(cmake/common.cmake)
include(cmake/license.cmake)
license_add_file("dense-gguf-server" "../LICENSE")
license_add_file("ggml authors (MIT)" "../build/licenses/LICENSE-llama.cpp")
license_add_file("Swagger UI" "vendor/swagger-ui/LICENSE")

set(vLicenseInstallDir "${CMAKE_INSTALL_DATAROOTDIR}/doc/dgs-server/licenses")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE"
  DESTINATION "${vLicenseInstallDir}" COMPONENT Licenses)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../build/licenses/"
  DESTINATION "${vLicenseInstallDir}" COMPONENT Licenses)
add_subdirectory(ggml)
add_subdirectory(src)
add_subdirectory(vendor)
add_subdirectory(common)
add_subdirectory(tools)

if(BUILD_SHARED_LIBS)
  install(TARGETS dgs dgs-common LIBRARY RUNTIME)
endif()
